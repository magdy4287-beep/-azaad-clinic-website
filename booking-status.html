const CLINIC_WEBSITE_URL = 'https://magdy4287-beep.github.io/-azaad-clinic-website/'; const CLINIC_SHARE_TEXT = '🌐 مشاركة الموقع الإلكتروني للعيادة عبر WhatsApp\n\n' + 'عيادة أزاد للصحة النفسية 🏥\n' + 'مساحة آمنة لبداية التغيير.\n\n' + 'يمكنك زيارة الموقع الإلكتروني للعيادة من هنا:\n' + CLINIC_WEBSITE_URL; function setupClinicWebsiteShare() { const shareButton = document.getElementById( 'shareClinicWhatsapp' ); if (!shareButton) { return; } const whatsappShareUrl = 'https://wa.me/?text=' + encodeURIComponent( CLINIC_SHARE_TEXT ); shareButton.href = whatsappShareUrl; shareButton.removeAttribute( 'target' ); shareButton.addEventListener( 'click', function () { shareButton.classList.add( 'is-opening' ); /* * ضغطة واحدة فقط: * Mobile → WhatsApp * Desktop → WhatsApp Web */ window.location.href = whatsappShareUrl; } ); }
